\begin{solutionbox}[Section B: Model Answers (Q11 to Q14)]
\textbf{Q.11 Storage Comparison}:
\begin{itemize}[leftmargin=*]
  \item \textbf{localStorage}: 5--10MB capacity, persistent indefinitely until cleared, never sent with HTTP requests.
  \item \textbf{sessionStorage}: 5MB capacity, scoped to tab lifetime (destroyed on tab close), never sent with HTTP.
  \item \textbf{Cookies}: 4KB capacity, configurable expiration date, automatically sent with every matching HTTP request header.
\end{itemize}

\textbf{Q.12 Box Model Calculation}:
$$\text{Rendered Box Width} = \text{width} + 2 \times \text{padding} + 2 \times \text{border} = 250 + (2 \times 15) + (2 \times 3) = 286\text{px}$$
$$\text{Total Space in Flow} = 286 + 2 \times \text{margin} = 286 + (2 \times 20) = 326\text{px}$$

\textbf{Q.13 Request-Response Lifecycle}:
1. DNS lookup resolves domain to IP address. 2. TCP 3-way handshake (SYN $\to$ SYN-ACK $\to$ ACK) + TLS handshake for HTTPS. 3. Browser sends HTTP GET request. 4. Server responds with 200 OK + HTML. 5. Browser executes Critical Rendering Path: DOM $\to$ CSSOM $\to$ Render Tree $\to$ Layout $\to$ Paint.

\textbf{Q.14 Scopes \& TDZ}:
\texttt{var} is function-scoped and hoisted to \texttt{undefined}. \texttt{let} and \texttt{const} are block-scoped. They are hoisted but remain in the Temporal Dead Zone (TDZ) where accessing them prior to declaration throws \texttt{ReferenceError}.
\end{solutionbox}

\begin{solutionbox}[Section C: Model Code for Q15 (Dynamic Cart Catalog)]
\begin{lstlisting}[language=HTML]
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Tech Catalog</title>
  <style>
    * { box-sizing: border-box; }
    body { font-family: Arial; display: flex; gap: 20px; padding: 20px; }
    #grid { flex: 2; display: grid; grid-template-columns: repeat(2, 1fr); gap: 15px; }
    .card { border: 1px solid #ccc; padding: 15px; border-radius: 6px; text-align: center; }
    #cart { flex: 1; border: 1px solid #333; padding: 15px; border-radius: 6px; }
  </style>
</head>
<body>
  <div id="grid"></div>
  <div id="cart">
    <h2>Shopping Cart</h2>
    <ul id="cartList"></ul>
    <h3 id="total">Total: &#8377;0</h3>
  </div>

  <script>
    const products = [
      { id: 1, name: "Wireless Mouse", price: 600 },
      { id: 2, name: "Mechanical Keyboard", price: 2200 },
      { id: 3, name: "HD Webcam", price: 1500 }
    ];
    const cart = [];

    function renderProducts() {
      const grid = document.getElementById("grid");
      products.forEach(p => {
        const card = document.createElement("div");
        card.className = "card";
        card.innerHTML = `<h3>${p.name}</h3><p>&#8377;${p.price}</p>`;
        const btn = document.createElement("button");
        btn.textContent = "Add to Cart";
        btn.onclick = () => { cart.push(p); updateCart(); };
        card.appendChild(btn);
        grid.appendChild(card);
      });
    }

    function updateCart() {
      const list = document.getElementById("cartList");
      list.innerHTML = "";
      cart.forEach(item => {
        const li = document.createElement("li");
        li.textContent = `${item.name} - &#8377;${item.price}`;
        list.appendChild(li);
      });
      const total = cart.reduce((sum, item) => sum + item.price, 0);
      document.getElementById("total").innerHTML = `Total: &#8377;${total}`;
    }

    renderProducts();
  </script>
</body>
</html>
\end{lstlisting}
\end{solutionbox}
