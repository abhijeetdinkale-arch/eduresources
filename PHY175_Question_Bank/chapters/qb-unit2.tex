\section{Section 3: Comprehensive Theory Questions \& Worked Solutions}
\begin{theoryq}{2.1}
State KCL and KVL. For a loaded voltage divider ($V_S, R_1, R_2, R_L$), derive output voltage.
\end{theoryq}

\begin{theorysol}
KCL: sum of currents at a node is zero. KVL: sum of voltages around a loop is zero. Loaded divider output: $V_{out} = V_S \frac{R_2 \parallel R_L}{R_1 + R_2 \parallel R_L} = V_S \frac{R_2 R_L}{R_1(R_2+R_L) + R_2 R_L}$.
\end{theorysol}

\begin{theoryq}{2.2}
Explain full-wave bridge rectifier with capacitor filter. Derive ripple voltage $V_{r(pp)} = \frac{I_{DC}}{2fC}$.
\end{theoryq}

\begin{theorysol}
Bridge rectifier conducts on both half-cycles ($f_r = 2f$). Capacitor charges to peak $V_m - 2V_D$ and discharges through load over $\Delta t = 1/(2f)$. Charge loss $\Delta Q = I_{DC}/(2f) = C V_{r(pp)} \implies V_{r(pp)} = \frac{I_{DC}}{2fC}$.
\end{theorysol}
