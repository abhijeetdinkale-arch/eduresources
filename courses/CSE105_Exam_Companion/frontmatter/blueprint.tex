\section*{Curriculum Blueprint \& Examination Specifications}

\begin{tcolorbox}[enhanced, colback=white, colframe=brandnavy, boxrule=1pt, arc=2mm, title={\Large\bfseries Course Outcomes (COs) \& Syllabus Mapping}]
\sffamily\small
\begin{description}[leftmargin=1.8em]
    \item[\textbf{CO1:}] Explain Linux OS layered architecture, kernel subsystems, file hierarchy (FHS), and multi-platform deployment modes.
    \item[\textbf{CO2:}] Apply core system utilities, command-line syntax, precision arithmetic, and man page documentation navigation.
    \item[\textbf{CO3:}] Implement POSIX permission security (octal/symbolic chmod), default masking (umask), ownership, and special bits (SUID/SGID/Sticky).
    \item[\textbf{CO4:}] Construct multi-stage stream filtering pipelines using wildcards, standard stream redirection, and regex engines (grep, cut, sort, uniq, tr).
    \item[\textbf{CO5:}] Administer multi-user environments, execute process lifecycle management, handle asynchronous signals, and edit documents via vi/vim.
    \item[\textbf{CO6:}] Develop modular, defensive Bash shell scripts and apply distributed version control workflows utilizing Git and modern CLI tools.
\end{description}
\end{tcolorbox}
