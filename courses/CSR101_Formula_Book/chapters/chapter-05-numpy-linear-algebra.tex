\section{NumPy Array Memory Layout \& Strides}
\begin{formulabox}{\texttt{ndarray} Internal Anatomy}
A NumPy array consists of a memory buffer and metadata:
\begin{itemize}[leftmargin=1.5em, itemsep=2pt]
    \item \textbf{Data Pointer:} Address of contiguous byte buffer in C heap.
    \item \textbf{Shape:} Tuple of dimension lengths $(d_0, d_1, \dots, d_{k-1})$.
    \item \textbf{Strides:} Tuple of byte offsets to advance one element along each dimension:
    \begin{equation}
    \text{Address}(i, j) = \text{base} + i \cdot s_0 + j \cdot s_1
    \end{equation}
    \item \textbf{Order:} C-contiguous (row-major: last index varies fastest) vs Fortran (column-major).
\end{itemize}
\end{formulabox}

\section{Broadcasting Rules Matrix (The Two Invariants)}
\begin{formulabox}{NumPy Broadcasting Conditions}
Two dimensions are compatible for element-wise operations if:
\begin{enumerate}[leftmargin=1.5em, itemsep=2pt]
    \item They are equal in size: $d_A = d_B$, OR
    \item One of them is $1$: $d_A = 1$ or $d_B = 1$ (the unit dimension is stretched without copying data).
\end{enumerate}
\textbf{Alignment Algorithm:}
\begin{itemize}[leftmargin=1em]
    \item Compare dimension shapes \textbf{from right to left} (trailing dimensions).
    \item Missing leading dimensions are prepended with size 1.
\end{itemize}
\textbf{Example Compatibility:}
\begin{align*}
\text{Array A: } & (256, 256, 3) \\
\text{Array B: } & (\phantom{256, 256, } 3) \quad \implies \text{Compatible! Result: } (256, 256, 3)
\end{align*}
\end{formulabox}
