\chapter{Diode Equations, Ratings \& Rectifier Parameters}

\section{Fundamental Semiconductor Equations}
\begin{formulabox}{Shockley Diode Equation \& Thermal Parameters}
\[ I_D = I_s \left( e^{\frac{V_D}{\eta V_T}} - 1 \right) \]
\begin{itemize}
    \item $I_s$: Reverse saturation current ($10^{-12}\text{ to } 10^{-9}\,\mathrm{A}$ for Silicon, $10^{-6}\,\mathrm{A}$ for Germanium).
    \item $\eta$: Ideality factor ($\eta = 1$ for modern planar Silicon at normal currents, $\eta = 2$ at very low currents).
    \item $V_T$: Thermal voltage:
    \[ V_T = \frac{k T}{q} = \frac{1.3806 \times 10^{-23} \times T}{1.6022 \times 10^{-19}} \approx \frac{T}{11600} \]
    At standard room temperature $T = 300\,\mathrm{K}$ ($27^\circ\mathrm{C}$):
    \[ V_T \approx 25.86\,\mathrm{mV} \approx 26\,\mathrm{mV} \]
\end{itemize}
\end{formulabox}

\section{Static and Dynamic Resistances}
\begin{formulabox}{Diode Resistances}
\begin{itemize}
    \item \textbf{Static (DC) Resistance}: Reciprocal of secant slope from origin:
    \[ R_D = \frac{V_D}{I_D} \]
    \item \textbf{Dynamic (AC / Incremental) Resistance}: Reciprocal of tangent slope at Q-point:
    \[ r_d = \frac{d V_D}{d I_D} = \frac{\eta V_T}{I_D + I_s} \approx \frac{\eta V_T}{I_D} = \frac{26\,\mathrm{mV}}{I_D} \quad (\text{at } 300\,\mathrm{K}, \eta=1) \]
\end{itemize}
\end{formulabox}
