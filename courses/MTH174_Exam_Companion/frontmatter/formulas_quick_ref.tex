\chapter*{High-Yield Formula Rapid-Reference}
\addcontentsline{toc}{chapter}{High-Yield Formula Rapid-Reference}

\begin{itemize}[leftmargin=1.5em]
    \item \textbf{Matrix Characteristic Polynomial ($3 \times 3$)}: $\lambda^3 - \operatorname{tr}(A)\lambda^2 + (\sum M_{ii})\lambda - \det(A) = 0$.
    \item \textbf{Cayley-Hamilton Identity}: $A^n + c_{n-1}A^{n-1} + \dots + c_1 A + c_0 I = 0 \implies A^{-1} = -\frac{1}{c_0}(A^{n-1} + \dots + c_1 I)$.
    \item \textbf{Differential Operator $D$ Inverse on $e^{ax}$}: $\frac{1}{f(D)}e^{ax} = \frac{1}{f(a)}e^{ax}$; if $f(a) = 0 \implies x\frac{1}{f'(D)}e^{ax}$.
    \item \textbf{Variation of Parameters}: $u(x) = -\int \frac{y_2 R}{W} dx$, $v(x) = \int \frac{y_1 R}{W} dx$, where $W = y_1 y_2' - y_2 y_1'$.
    \item \textbf{Fourier Coefficients on $[-\pi, \pi]$}: $a_0 = \frac{1}{\pi}\int_{-\pi}^\pi f(x)dx$, $a_n = \frac{1}{\pi}\int_{-\pi}^\pi f(x)\cos nx dx$, $b_n = \frac{1}{\pi}\int_{-\pi}^\pi f(x)\sin nx dx$.
    \item \textbf{Jacobian Identity}: $\frac{\partial(u,v)}{\partial(x,y)} \cdot \frac{\partial(x,y)}{\partial(u,v)} = 1$.
    \item \textbf{Double Integral Polar Area Element}: $dx \, dy = r \, dr \, d\theta$.
\end{itemize}
