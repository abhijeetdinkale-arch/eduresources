\begin{enumerate}[label=\textbf{\arabic*.}]
    \item If $A$ is an orthogonal matrix, then $\det(A)$ is:
    \begin{enumerate*}[label=(\alph*)]
        \item $0$ \quad
        \item $\pm 1$ \quad
        \item $\pm 2$ \quad
        \item Any real number
    \end{enumerate*}
    \item The eigenvalues of a real symmetric matrix are always:
    \begin{enumerate*}[label=(\alph*)]
        \item Real \quad
        \item Purely imaginary \quad
        \item Complex with positive real part \quad
        \item Zero
    \end{enumerate*}
    \item The complementary function of $(D^2 + 4)y = \sin 2x$ is:
    \begin{enumerate*}[label=(\alph*)]
        \item $c_1 \cos 2x + c_2 \sin 2x$ \quad
        \item $c_1 e^{2x} + c_2 e^{-2x}$ \quad
        \item $(c_1 + c_2 x)e^{2x}$ \quad
        \item $c_1 \cos 4x$
    \end{enumerate*}
\end{enumerate}
