\subsection*{Part B: Descriptive Analytical Section (40 Marks)}

\questionitem{1}{
    Solve $(D^2 + 1)y = \sin x \sin 2x$.
}{10}{CO3}{L3 Apply}

\questionitem{2}{
    Obtain the Fourier expansion for $f(x) = x \sin x$ on $[-\pi, \pi]$.
}{10}{CO4}{L4 Analyze}

\questionitem{3}{
    Investigate the maxima and minima of $u(x,y,z) = x^2 + y^2 + z^2$ subject to $ax + by + cz = p$.
}{10}{CO5}{L4 Analyze}

\questionitem{4}{
    Evaluate $\iint_R x y dx dy$ over the region bounded by $y = x^2$ and $y = x$.
}{10}{CO6}{L3 Apply}

\questionitem{5}{
    Evaluate $\iiint_V (x + y + z) dxdydz$ over the tetrahedron bounded by $x \ge 0, y \ge 0, z \ge 0$ and $x + y + z \le 1$.
}{10}{CO6}{L4 Analyze}
