\subsection*{Solutions and Marking Rubrics --- End-Term Set A}

\begin{solutionbox}[Solution to Question 2(a) --- Variation of Parameters]
Complementary function: $m^2 - 2m + 1 = 0 \implies m = 1, 1$.
$y_c = (c_1 + c_2 x)e^x \implies y_1 = e^x, \; y_2 = x e^x$.
Wronskian:
\[
W = y_1 y_2' - y_2 y_1' = e^x(e^x + x e^x) - (x e^x)(e^x) = e^{2x}
\]
With $R(x) = \frac{e^x}{x}$:
\[
u(x) = -\int \frac{y_2 R}{W} dx = -\int \frac{(x e^x)(e^x/x)}{e^{2x}} dx = -\int 1 dx = -x
\]
\[
v(x) = \int \frac{y_1 R}{W} dx = \int \frac{(e^x)(e^x/x)}{e^{2x}} dx = \int \frac{1}{x} dx = \ln|x|
\]
Particular integral:
\[
y_p = -x e^x + (x e^x)\ln|x|
\]
General solution:
\[
\mathbf{y(x) = (c_1 + c_2 x)e^x - x e^x + x e^x \ln|x| = (c_1 + c_3 x)e^x + x e^x \ln|x|}
\]
\end{solutionbox}

\begin{solutionbox}[Solution to Question 4(b) --- Change of Order]
Region bounded by $x = y^2 \implies y = \sqrt{x}$ and $x = 1$, with $y \in [0, 1]$.
Switching to vertical strips: $x \in [0, 1]$, $y \in [0, \sqrt{x}]$.
\[
I = \int_0^1 \left( \int_0^{\sqrt{x}} y e^{x^2} dy \right) dx = \int_0^1 e^{x^2} \left[ \frac{y^2}{2} \right]_0^{\sqrt{x}} dx = \frac{1}{2}\int_0^1 x e^{x^2} dx = \frac{1}{4}[e^{x^2}]_0^1 = \mathbf{\frac{e - 1}{4}}
\]
\end{solutionbox}
