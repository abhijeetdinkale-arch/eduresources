\chapter{Unit 4: Fourier Series Formulas \& Expansions}

\begin{formulabox}{Euler-Fourier Formulas on General Intervals}
\begin{center}
\begin{tabularx}{\textwidth}{l X X}
\toprule
\textbf{Interval} & \textbf{Expansion Form} & \textbf{Coefficients} \\
\midrule
$[-\pi, \pi]$ & $\frac{a_0}{2} + \sum_{n=1}^\infty (a_n \cos nx + b_n \sin nx)$ & $a_0 = \frac{1}{\pi}\int_{-\pi}^\pi f(x)dx$ \newline $a_n = \frac{1}{\pi}\int_{-\pi}^\pi f(x)\cos nx dx$ \newline $b_n = \frac{1}{\pi}\int_{-\pi}^\pi f(x)\sin nx dx$ \\
\addlinespace
$[0, 2\pi]$ & $\frac{a_0}{2} + \sum_{n=1}^\infty (a_n \cos nx + b_n \sin nx)$ & Same formulas with limits $0$ to $2\pi$ \\
\addlinespace
$[-L, L]$ & $\frac{a_0}{2} + \sum_{n=1}^\infty \left[ a_n \cos\frac{n\pi x}{L} + b_n \sin\frac{n\pi x}{L} \right]$ & $a_0 = \frac{1}{L}\int_{-L}^L f(x)dx$ \newline $a_n = \frac{1}{L}\int_{-L}^L f(x)\cos\frac{n\pi x}{L} dx$ \newline $b_n = \frac{1}{L}\int_{-L}^L f(x)\sin\frac{n\pi x}{L} dx$ \\
\bottomrule
\end{tabularx}
\end{center}
\end{formulabox}

\begin{formulabox}{Half-Range Series on $[0, L]$}
\begin{itemize}
    \item \textbf{Half-Range Cosine Series}:
    \begin{equation*}
    f(x) = \frac{a_0}{2} + \sum_{n=1}^\infty a_n \cos\left(\frac{n\pi x}{L}\right), \quad a_0 = \frac{2}{L}\int_0^L f(x)dx, \quad a_n = \frac{2}{L}\int_0^L f(x)\cos\left(\frac{n\pi x}{L}\right)dx
    \end{equation*}
    \item \textbf{Half-Range Sine Series}:
    \begin{equation*}
    f(x) = \sum_{n=1}^\infty b_n \sin\left(\frac{n\pi x}{L}\right), \quad b_n = \frac{2}{L}\int_0^L f(x)\sin\left(\frac{n\pi x}{L}\right)dx
    \end{equation*}
\end{itemize}
\end{formulabox}

\begin{formulabox}{Helpful Trigonometric Values at $n \in \mathbb{Z}$}
$\cos(n\pi) = (-1)^n$, $\sin(n\pi) = 0$, $\sin\left(\frac{n\pi}{2}\right) = \begin{cases} 0, & n \text{ even} \\ (-1)^{(n-1)/2}, & n \text{ odd} \end{cases}$, $\cos\left(\frac{n\pi}{2}\right) = \begin{cases} (-1)^{n/2}, & n \text{ even} \\ 0, & n \text{ odd} \end{cases}$.
\end{formulabox}
