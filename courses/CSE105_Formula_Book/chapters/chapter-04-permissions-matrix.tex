\begin{cheatbox}{The Octal Permission Calculation Matrix}
Every 3-character permission triplet is calculated as: $\text{Triad Value} = (r \times 4) + (w \times 2) + (x \times 1)$.
\begin{center}
\begin{tabular}{c c c c c}
\toprule
\textbf{Octal Digit} & \textbf{Binary Bits} & \textbf{Symbolic} & \textbf{Regular File Meaning} & \textbf{Directory Meaning} \\
\midrule
\texttt{7} & \texttt{111} & \texttt{rwx} & Full read, write, and execute & List, modify files, and enter directory \\
\texttt{6} & \texttt{110} & \texttt{rw-} & Read and write (no execution) & List and modify (cannot traverse) \\
\texttt{5} & \texttt{101} & \texttt{r-x} & Read and execute (programs) & List and traverse (cannot alter files) \\
\texttt{4} & \texttt{100} & \texttt{r--} & Read-only document & List file names only (cannot access inodes) \\
\texttt{0} & \texttt{000} & \texttt{---} & Completely restricted (no access) & Zero access \\
\bottomrule
\end{tabular}
\end{center}
\end{cheatbox}

\begin{cheatbox}{The umask Calculation Formulas}
\begin{equation}
\text{Default File Mode} = \text{Base File (666)} - \text{umask} \quad (\text{Execution bit is NEVER granted by default})
\end{equation}
\begin{equation}
\text{Default Directory Mode} = \text{Base Directory (777)} - \text{umask}
\end{equation}
\begin{itemize}[leftmargin=1.5em]
    \item \texttt{umask 022} $\to$ Files: $666 - 022 = \mathbf{644}$ (\texttt{rw-r--r--}); Directories: $777 - 022 = \mathbf{755}$ (\texttt{rwxr-xr-x}).
    \item \texttt{umask 027} $\to$ Files: $666 - 027 = \mathbf{640}$ (\texttt{rw-r-----}); Directories: $777 - 027 = \mathbf{750}$ (\texttt{rwxr-x---}).
    \item \texttt{umask 077} $\to$ Files: $666 - 077 = \mathbf{600}$ (\texttt{rw-------}); Directories: $777 - 077 = \mathbf{700}$ (\texttt{rwx------}).
\end{itemize}
\end{cheatbox}
