\begin{cheatbox}{Terminal Job Control Commands}
\begin{itemize}[leftmargin=1.5em]
    \item \texttt{cmd \&}: Spawns command directly into the background.
    \item \texttt{jobs -l}: Lists active shell jobs with job numbers and PIDs.
    \item \texttt{Ctrl+Z}: Sends \texttt{SIGTSTP}, stopping the active foreground job.
    \item \texttt{bg \%N}: Resumes stopped job $N$ running in the background.
    \item \texttt{fg \%N}: Restores background job $N$ into the active foreground terminal.
    \item \texttt{nohup cmd \&}: Detaches command from terminal; ignores \texttt{SIGHUP}.
\end{itemize}
\end{cheatbox}
