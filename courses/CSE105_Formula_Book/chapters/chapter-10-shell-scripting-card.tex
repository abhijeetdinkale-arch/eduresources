\begin{codebox}[Core Control Structures Skeleton Templates]
# 1. if-then-elif-else construct:
if [ "$ENV" = "production" ] && [ -f "$CONF" ]; then
    echo "Starting in Production mode..."
elif [ "$ENV" = "staging" ]; then
    echo "Starting in Staging mode..."
else
    echo "Defaulting to Development..."
fi

# 2. case-esac construct:
case "$1" in
    start) systemctl start app ;;
    stop)  systemctl stop app ;;
    *)     echo "Usage: $0 {start|stop}" ; exit 1 ;;
esac

# 3. for loop over collections:
for server in web01 web02 db01; do
    ping -c 1 "$server" >/dev/null && echo "$server is online"
done

# 4. while loop reading files line-by-line safely:
while IFS= read -r line || [ -n "$line" ]; do
    echo "Record: $line"
done < data.csv
\end{codebox}

\begin{tipbox}
Always include \texttt{set -euo pipefail} at the top of production Bash scripts. It causes scripts to exit immediately on errors (\texttt{-e}), fail on uninitialized variables (\texttt{-u}), and fail on pipeline errors (\texttt{-o pipefail}).
\end{tipbox}
