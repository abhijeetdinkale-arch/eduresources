\begin{cheatbox}{Stream Processing Utilities: cut, sort, uniq, tr, wc}
\begin{itemize}[leftmargin=1.5em]
    \item \texttt{cut -d":" -f1,3}: Extracts colon-separated fields 1 and 3.
    \item \texttt{sort -n -r -k2 -t":"}: Sorts numerically (\texttt{-n}), descending (\texttt{-r}), on field 2 (\texttt{-k2}), using delimiter \texttt{:} (\texttt{-t":"}).
    \item \texttt{uniq -c}: Counts occurrences of adjacent identical lines (requires prior \texttt{sort}!).
    \item \texttt{tr 'a-z' 'A-Z'}: Translates lowercase letters to uppercase.
    \item \texttt{tr -d '\textbackslash r'}: Deletes Windows carriage return characters.
    \item \texttt{tr -s ' '}: Squeezes repeated spaces into a single space.
    \item \texttt{wc -l}: Counts total lines in file or stream.
\end{itemize}
\end{cheatbox}

\begin{codebox}[Top 5 High-Yield Filter One-Liners]
# 1. Top 10 IP addresses accessing a web server:
awk '{print $1}' access.log | sort | uniq -c | sort -rn | head -n 10

# 2. Extract all local system usernames with UID >= 1000:
awk -F: '$3 >= 1000 {print $1, $3}' /etc/passwd

# 3. Strip all comment and blank lines from a configuration file:
grep -vE "^(#|$)" /etc/nginx/nginx.conf

# 4. Find all files modified in the past 24 hours:
find . -type f -mtime -1

# 5. Count total words in text while ignoring punctuation:
tr -s '[:punct:][:space:]' '\n' < doc.txt | tr '[:upper:]' '[:lower:]' | sort | uniq -c | sort -rn
\end{codebox}
