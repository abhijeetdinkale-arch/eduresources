\section{SI Electrical Units and Dimensional Equivalences}

\begin{formulabox}{Derived Electrical Quantities}
\begin{itemize}[leftmargin=1.5em]
    \item \textbf{Resistance} ($\Omega$): $1\,\Omega = 1\text{ V/A} = 1\text{ kg}\cdot\text{m}^2/(\text{s}^3\cdot\text{A}^2)$.
    \item \textbf{Conductance} (S): $1\text{ S} = 1\,\Omega^{-1} = 1\text{ A/V}$.
    \item \textbf{Capacitance} (F): $1\text{ F} = 1\text{ C/V} = 1\text{ A}\cdot\text{s/V} = 1\text{ s}/\Omega$.
    \item \textbf{Inductance} (H): $1\text{ H} = 1\text{ V}\cdot\text{s/A} = 1\,\Omega\cdot\text{s} = 1\text{ Wb/A}$.
    \item \textbf{Magnetic Flux} (Wb): $1\text{ Wb} = 1\text{ V}\cdot\text{s} = 1\text{ T}\cdot\text{m}^2$.
    \item \textbf{Magnetic Flux Density} (T): $1\text{ T} = 1\text{ Wb/m}^2 = 1\text{ N}/(\text{A}\cdot\text{m}) = 10^4\text{ Gauss}$.
\end{itemize}
\end{formulabox}
