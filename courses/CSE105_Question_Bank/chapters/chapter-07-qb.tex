\section{Technical Viva-Voce Questions \& Conceptual Answers}

\subsection*{Q1. Explain the 7 fields of /etc/passwd with an exact example entry.}
\begin{vivasol}
Entry: \texttt{student:x:1001:1001:CS Undergrad,Lab 4:/home/student:/bin/bash}
\begin{enumerate}[leftmargin=1.5em]
    \item \texttt{student}: Username / login name.
    \item \texttt{x}: Password placeholder indicating hash is in \texttt{/etc/shadow}.
    \item \texttt{1001}: User ID (UID).
    \item \texttt{1001}: Primary Group ID (GID).
    \item \texttt{CS Undergrad,Lab 4}: GECOS comment field describing user.
    \item \texttt{/home/student}: Absolute path to home directory.
    \item \texttt{/bin/bash}: Login shell program spawned upon authentication.
\end{enumerate}
\end{vivasol}
