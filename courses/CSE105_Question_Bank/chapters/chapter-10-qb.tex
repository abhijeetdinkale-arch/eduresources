\section{Shell Scripting \& DevOps Automation Challenges}

\begin{scenariobox}{10.1 --- Automated Server Log Archival Engine}
\textbf{Scenario:} Design a complete, production-grade Bash script that parses a log directory, compresses files older than 7 days into a dated tarball, deletes original logs upon successful compression, and logs all operations with timestamps.
\begin{scriptlisting}
#!/bin/bash
set -euo pipefail

LOG_DIR="/var/log/app"
BACKUP_DIR="/backup/logs"
DATE_STAMP=$(date +%Y%m%d_%H%M%S)
ARCHIVE_NAME="logs_archive_${DATE_STAMP}.tar.gz"

# 1. Input Validation
if [ ! -d "$LOG_DIR" ]; then
    echo "[ERROR] Log directory $LOG_DIR does not exist." >&2
    exit 1
fi
mkdir -p "$BACKUP_DIR"

# 2. Identify and Archive Old Logs
OLD_FILES=$(find "$LOG_DIR" -type f -name "*.log" -mtime +7)

if [ -z "$OLD_FILES" ]; then
    echo "[INFO] No log files older than 7 days identified. Exiting."
    exit 0
fi

echo "[INFO] Commencing archival into $BACKUP_DIR/$ARCHIVE_NAME..."
tar -czf "$BACKUP_DIR/$ARCHIVE_NAME" -T <(echo "$OLD_FILES")

# 3. Verify Archive Integrity Before Deletion
if [ -s "$BACKUP_DIR/$ARCHIVE_NAME" ]; then
    echo "[INFO] Archive generated successfully. Removing archived logs..."
    echo "$OLD_FILES" | xargs rm -f
    echo "[SUCCESS] Archival cycle concluded successfully."
else
    echo "[FATAL] Archive file was empty! Aborting deletion." >&2
    exit 2
fi
\end{scriptlisting}
\end{scenariobox}
