\subsection*{Core Quantitative Formulations}
\begin{itemize}[leftmargin=1.2em]
    \item \textbf{IPAT Identity}: $I = P \times A \times T$
    \item \textbf{Ghyben-Herzberg}: $z = 40 h$
    \item \textbf{USLE}: $A = R \cdot K \cdot LS \cdot C \cdot P$
    \item \textbf{Lindeman 10\%}: $\lambda = \frac{E_n}{E_{n-1}} \times 100\% \approx 10\%$
    \item \textbf{NPP}: $\text{NPP} = \text{GPP} - R_a$
    \item \textbf{Shannon Index}: $H' = -\sum p_i \ln p_i$
    \item \textbf{Pielou Evenness}: $J' = H' / \ln S$
    \item \textbf{Simpson Index}: $D = \sum p_i^2$, Diversity $= 1 - D$
    \item \textbf{BOD Kinetics}: $y(t) = L_0(1 - e^{-k_d t})$
    \item \textbf{Streeter-Phelps}: $D(t) = \frac{k_d L_0}{k_r - k_d}(e^{-k_d t} - e^{-k_r t}) + D_0 e^{-k_r t}$
    \item \textbf{Disaster Risk}: $\text{Risk} = \frac{H \times E \times V}{C}$
    \item \textbf{Earthquake Energy}: $\log_{10} E = 4.8 + 1.5 M$
    \item \textbf{Rule of 70 (Doubling)}: $T_d \approx \frac{70}{r\%}$
    \item \textbf{Logistic Growth}: $\frac{dN}{dt} = rN\left(1 - \frac{N}{K}\right)$
\end{itemize}
