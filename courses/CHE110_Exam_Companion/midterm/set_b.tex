\section{Mid-Term Examination --- Set B (30 Marks)}
\label{sec:midterm_set_b}

\LPUPaperHeader{Mid-Term Examination}{B}{90 Minutes}{30 Marks}{CO1 (Intro & Sustainability), CO2 (Resources & Ecosystems), CO3 (Biodiversity)}

\begin{center}
\sffamily\textbf{\color{AlertRed}Note: 30 Multiple Choice Questions. Each question carries 1 mark. Negative marking $-0.25$ marks for wrong answers.}
\end{center}

\begin{enumerate}[label=\textbf{\arabic*.}]
    \item {What is one important aim of environmental studies?}
    \begin{enumerate*}[label=(\alph*)]
        \item {To increase resource wastage} \quad
        \item {To discourage public awareness} \quad
        \item {To promote environmental protection} \quad
        \item {To remove all natural habitats}
    \end{enumerate*}
    \item {Environmental studies is important because it helps people:}
    \begin{enumerate*}[label=(\alph*)]
        \item {Understand environmental problems} \quad
        \item {Use resources without limits} \quad
        \item {Ignore pollution problems} \quad
        \item {Avoid learning about ecosystems}
    \end{enumerate*}
    \item {What is the basic meaning of sustainability?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Avoiding every form of development} \quad
        \item {Increasing waste in natural systems} \quad
        \item {Using resources responsibly over time} \quad
        \item {Using all resources immediately}
    \end{enumerate*}
    \item {Sustainable development aims to meet present needs without:}
    \begin{enumerate*}[label=(\alph*)]
        \item {Harming future generations' needs} \quad
        \item {Improving present living conditions} \quad
        \item {Using any natural resources} \quad
        \item {Supporting economic activities}
    \end{enumerate*}
    \item {Which action supports sustainable development?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Dumping waste into rivers} \quad
        \item {Using renewable energy} \quad
        \item {Wasting clean drinking water} \quad
        \item {Cutting forests without replanting}
    \end{enumerate*}
    \item {What is carrying capacity?}
    \begin{enumerate*}[label=(\alph*)]
        \item {The speed of population growth} \quad
        \item {The amount of rainfall in a region} \quad
        \item {The total number of species on Earth} \quad
        \item {The largest population an environment can support}
    \end{enumerate*}
    \item {What may happen when a population exceeds its carrying capacity?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Competition between organisms ends} \quad
        \item {Resources automatically increase} \quad
        \item {Pollution always disappears} \quad
        \item {Resources may become depleted}
    \end{enumerate*}
    \item {Which of the following is commonly considered a pillar of sustainability?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Entertainment} \quad
        \item {Fashion} \quad
        \item {Advertising} \quad
        \item {Environment}
    \end{enumerate*}
    \item {Which set correctly lists the three main pillars of sustainability?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Energy, transport, and tourism} \quad
        \item {Environment, economy, and society} \quad
        \item {Science, language, and history} \quad
        \item {Agriculture, industry, and trade}
    \end{enumerate*}
    \item {How many Sustainable Development Goals were adopted by the United Nations?}
    \begin{enumerate*}[label=(\alph*)]
        \item {10} \quad
        \item {17} \quad
        \item {twenty} \quad
        \item {15}
    \end{enumerate*}
    \item {Which action helps conserve energy at home?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Keeping doors open while cooling} \quad
        \item {Running empty washing machines} \quad
        \item {Switching off unused lights} \quad
        \item {Using appliances throughout the day}
    \end{enumerate*}
    \item {Which of the following is a biotic component of an ecosystem?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Sunlight} \quad
        \item {Water} \quad
        \item {Soil} \quad
        \item {Plant}
    \end{enumerate*}
    \item {What is the main role of decomposers in an ecosystem?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Breaking down dead matter} \quad
        \item {Hunting living herbivores} \quad
        \item {Preventing rainfall from occurring} \quad
        \item {Producing energy from sunlight}
    \end{enumerate*}
    \item {Which of the following is an aquatic ecosystem?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Grassland} \quad
        \item {Pond} \quad
        \item {Forest} \quad
        \item {Desert}
    \end{enumerate*}
    \item {Which ecosystem usually receives very little rainfall?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Rainforest} \quad
        \item {Wetland} \quad
        \item {Estuary} \quad
        \item {Desert}
    \end{enumerate*}
    \item {What is the primary source of energy for most ecosystems?}
    \begin{enumerate*}[label=(\alph*)]
        \item {The Sun} \quad
        \item {The wind} \quad
        \item {The Moon} \quad
        \item {The soil}
    \end{enumerate*}
    \item {In the food chain grass $\to$ deer $\to$ tiger , which organism is the producer?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Tiger} \quad
        \item {Grass} \quad
        \item {None} \quad
        \item {Deer}
    \end{enumerate*}
    \item {What does a food web represent?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Interconnected food chains} \quad
        \item {Only decomposer activity} \quad
        \item {Only plant relationships} \quad
        \item {A single feeding pathway}
    \end{enumerate*}
    \item {Which organisms usually form the base of an ecological pyramid?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Decomposers} \quad
        \item {Herbivores} \quad
        \item {Producers} \quad
        \item {Carnivores}
    \end{enumerate*}
    \item {What is ecological succession?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Sudden transfer of energy between organisms} \quad
        \item {Daily movement of animals between habitats} \quad
        \item {Seasonal change in local weather conditions} \quad
        \item {Gradual change in a biological community}
    \end{enumerate*}
    \item {What is an invasive alien species?}
    \begin{enumerate*}[label=(\alph*)]
        \item {A migratory species that returns yearly} \quad
        \item {A domestic species raised by people} \quad
        \item {A native species with a small population} \quad
        \item {A non-native species that causes harm}
    \end{enumerate*}
    \item {Which situation is an example of human-wildlife conflict?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Deer grazing within a national park} \quad
        \item {Elephants damaging crops near a forest} \quad
        \item {Butterflies pollinating flowers in a garden} \quad
        \item {Birds nesting inside a protected wetland}
    \end{enumerate*}
    \item {When is a species considered extinct?}
    \begin{enumerate*}[label=(\alph*)]
        \item {When no living individual remains} \quad
        \item {When it moves to another habitat} \quad
        \item {When it lives in one small region} \quad
        \item {When its population begins decreasing}
    \end{enumerate*}
    \item {An endangered species is one that:}
    \begin{enumerate*}[label=(\alph*)]
        \item {Has recently entered a new habitat} \quad
        \item {Faces a very high risk of extinction} \quad
        \item {Occurs commonly across many regions} \quad
        \item {Is maintained mainly as a farm animal}
    \end{enumerate*}
    \item {A vulnerable species is best described as a species that:}
    \begin{enumerate*}[label=(\alph*)]
        \item {Faces a high risk of extinction} \quad
        \item {Exists only under human care} \quad
        \item {Occurs abundantly in every habitat} \quad
        \item {Has already disappeared completely}
    \end{enumerate*}
    \item {Which feature is commonly associated with a rare species?}
    \begin{enumerate*}[label=(\alph*)]
        \item {A complete absence of living individuals} \quad
        \item {A rapid increase in every ecosystem} \quad
        \item {A small population or limited distribution} \quad
        \item {A large population across many continents}
    \end{enumerate*}
    \item {What does EDGE stand for in biodiversity conservation?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Evolutionarily Distinct and Globally Endangered} \quad
        \item {Evolutionarily Dominant and Globally Endemic} \quad
        \item {Environmentally Dependent and Genetically Enhanced} \quad
        \item {Ecologically Diverse and Geographically Established}
    \end{enumerate*}
    \item {Which is an example of in-situ conservation?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Breeding tigers in a zoological park} \quad
        \item {Maintaining tissues in a laboratory} \quad
        \item {Storing plant seeds in a seed bank} \quad
        \item {Protecting tigers in a national park}
    \end{enumerate*}
    \item {Which is an example of ex-situ conservation?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Protecting trees in a biosphere reserve} \quad
        \item {Preserving plants in a sacred grove} \quad
        \item {Conserving animals in a wildlife sanctuary} \quad
        \item {Storing seeds in a seed bank}
    \end{enumerate*}
    \item {How many major biogeographic zones are commonly recognized in India?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Ten} \quad
        \item {Fourteen} \quad
        \item {Twelve} \quad
        \item {Eight}
    \end{enumerate*}
\end{enumerate}
