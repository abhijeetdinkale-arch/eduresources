\section{Mid-Term Examination --- Set A (30 Marks)}
\label{sec:midterm_set_a}

\LPUPaperHeader{Mid-Term Examination}{A}{90 Minutes}{30 Marks}{CO1 (Intro & Sustainability), CO2 (Resources & Ecosystems), CO3 (Biodiversity)}

\begin{center}
\sffamily\textbf{\color{AlertRed}Note: 30 Multiple Choice Questions. Each question carries 1 mark. Negative marking $-0.25$ marks for wrong answers.}
\end{center}

\begin{enumerate}[label=\textbf{\arabic*.}]
    \item {What does the term environment refer to?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Only the weather of a place} \quad
        \item {The surroundings in which organisms live} \quad
        \item {Only the plants in a forest} \quad
        \item {Only the buildings around people}
    \end{enumerate*}
    \item {Which of the following is a natural component of the environment?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Air} \quad
        \item {Bridge} \quad
        \item {Factory} \quad
        \item {Road}
    \end{enumerate*}
    \item {Which component of the environment includes plants, animals, and microorganisms?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Abiotic component} \quad
        \item {Geological component} \quad
        \item {Physical component} \quad
        \item {Biotic component}
    \end{enumerate*}
    \item {Which of the following is an abiotic component of the environment?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Trees} \quad
        \item {Birds} \quad
        \item {Water} \quad
        \item {Bacteria}
    \end{enumerate*}
    \item {Which sphere of Earth consists mainly of air?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Atmosphere} \quad
        \item {Hydrosphere} \quad
        \item {Biosphere} \quad
        \item {Lithosphere}
    \end{enumerate*}
    \item {The hydrosphere includes which of the following?}
    \begin{enumerate*}[label=(\alph*)]
        \item {All rocks on Earth} \quad
        \item {All living organisms} \quad
        \item {All water on Earth} \quad
        \item {All gases around Earth}
    \end{enumerate*}
    \item {Which sphere includes the solid outer part of Earth?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Lithosphere} \quad
        \item {Atmosphere} \quad
        \item {Hydrosphere} \quad
        \item {Biosphere}
    \end{enumerate*}
    \item {The biosphere is the part of Earth where:}
    \begin{enumerate*}[label=(\alph*)]
        \item {Life exists} \quad
        \item {Only rocks occur} \quad
        \item {Only clouds are formed} \quad
        \item {Only water is stored}
    \end{enumerate*}
    \item {Why is environmental studies called a multidisciplinary subject?}
    \begin{enumerate*}[label=(\alph*)]
        \item {It excludes social and economic issues} \quad
        \item {It studies only one type of organism} \quad
        \item {It uses knowledge from many subjects} \quad
        \item {It focuses only on chemical reactions}
    \end{enumerate*}
    \item {Which field helps environmental studies understand human population and communities?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Sociology} \quad
        \item {Trigonometry} \quad
        \item {Linguistics} \quad
        \item {Astronomy}
    \end{enumerate*}
    \item {What are natural resources?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Goods imported from other countries} \quad
        \item {Products made only in factories} \quad
        \item {Materials obtained from nature} \quad
        \item {Objects created by computer programs}
    \end{enumerate*}
    \item {Which of the following is a renewable resource?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Solar energy} \quad
        \item {Coal} \quad
        \item {Petroleum} \quad
        \item {Natural gas}
    \end{enumerate*}
    \item {Which resource is classified as non-renewable?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Rainwater} \quad
        \item {Wind} \quad
        \item {Sunlight} \quad
        \item {Coal}
    \end{enumerate*}
    \item {Which resource provides habitat for many wild plants and animals?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Petroleum} \quad
        \item {Mineral ore} \quad
        \item {Natural gas} \quad
        \item {Forest}
    \end{enumerate*}
    \item {Which activity uses the largest share of freshwater in many countries?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Agriculture} \quad
        \item {Communication} \quad
        \item {Transportation} \quad
        \item {Construction}
    \end{enumerate*}
    \item {Which energy resource is obtained from moving air?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Tidal energy} \quad
        \item {Geothermal energy} \quad
        \item {Wind energy} \quad
        \item {Coal energy}
    \end{enumerate*}
    \item {Which method helps reduce soil erosion on steep slopes?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Overgrazing} \quad
        \item {Deforestation} \quad
        \item {Terrace farming} \quad
        \item {Open mining}
    \end{enumerate*}
    \item {What is a major result of excessive removal of trees?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Afforestation} \quad
        \item {Water harvesting} \quad
        \item {Deforestation} \quad
        \item {Soil formation}
    \end{enumerate*}
    \item {Which practice helps replenish groundwater?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Rainwater harvesting} \quad
        \item {Clearing vegetation} \quad
        \item {Burning fossil fuels} \quad
        \item {Paving open land}
    \end{enumerate*}
    \item {Which individual action directly conserves water?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Using disposable plates} \quad
        \item {Burning household waste} \quad
        \item {Repairing leaking taps} \quad
        \item {Leaving lights switched on}
    \end{enumerate*}
    \item {What does biodiversity refer to?}
    \begin{enumerate*}[label=(\alph*)]
        \item {The variety of living organisms} \quad
        \item {The study of weather patterns} \quad
        \item {The distribution of mineral resources} \quad
        \item {The movement of tectonic plates}
    \end{enumerate*}
    \item {Which level of biodiversity refers to variations in genes within a species?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Landscape diversity} \quad
        \item {Species diversity} \quad
        \item {Genetic diversity} \quad
        \item {Ecosystem diversity}
    \end{enumerate*}
    \item {The variety of different species present in an area is called:}
    \begin{enumerate*}[label=(\alph*)]
        \item {Ecosystem diversity} \quad
        \item {Species diversity} \quad
        \item {Genetic diversity} \quad
        \item {Cultural diversity}
    \end{enumerate*}
    \item {Forests, grasslands, deserts, and wetlands found in a region demonstrate:}
    \begin{enumerate*}[label=(\alph*)]
        \item {Ecosystem diversity} \quad
        \item {Population diversity} \quad
        \item {Species diversity} \quad
        \item {Genetic diversity}
    \end{enumerate*}
    \item {Which of the following is an ecosystem service provided by bees?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Pollination of crops} \quad
        \item {Production of plastics} \quad
        \item {Formation of rocks} \quad
        \item {Extraction of metals}
    \end{enumerate*}
    \item {The use of plants to develop medicines is an example of biodiversity's:}
    \begin{enumerate*}[label=(\alph*)]
        \item {Climatic value} \quad
        \item {Astronomical value} \quad
        \item {Economic value} \quad
        \item {Tectonic value}
    \end{enumerate*}
    \item {Enjoying the beauty of flowers and wildlife represents which value of biodiversity?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Economic value} \quad
        \item {Informational value} \quad
        \item {Ecological value} \quad
        \item {Aesthetic value}
    \end{enumerate*}
    \item {What is a biodiversity hotspot?}
    \begin{enumerate*}[label=(\alph*)]
        \item {A region used mainly for intensive agriculture} \quad
        \item {A region receiving very little rainfall} \quad
        \item {A region containing many active volcanoes} \quad
        \item {A region rich in endemic species and under threat}
    \end{enumerate*}
    \item {Which activity is a direct cause of habitat loss?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Clearing forests for farms} \quad
        \item {Restoring degraded wetlands} \quad
        \item {Planting native tree species} \quad
        \item {Creating wildlife corridors}
    \end{enumerate*}
    \item {What is wildlife poaching?}
    \begin{enumerate*}[label=(\alph*)]
        \item {Licensed treatment of injured animals} \quad
        \item {Illegal hunting or capture of wild animals} \quad
        \item {Seasonal movement of wild animals} \quad
        \item {Scientific observation of wild animals}
    \end{enumerate*}
\end{enumerate}
