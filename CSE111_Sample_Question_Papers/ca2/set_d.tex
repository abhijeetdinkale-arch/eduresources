\begin{expbox}[Technical Solutions --- CA-2 Set D]
\begin{itemize}[leftmargin=5mm]
    \item \textbf{Q.1 (C):} In Hexadecimal, $\text{A}=10, \text{B}=11, \text{C}=12, \text{D}=13$.
    \item \textbf{Q.5 (A):} $111010_2$ grouped in threes: $\underbrace{111}_{7}\;\underbrace{010}_{2} = (72)_8$.
    \item \textbf{Q.8 (B):} $0.11_2 = 1(0.5) + 1(0.25) = 0.75_{10}$.
    \item \textbf{Q.11 (B):} $(2\text{B})_{16} = 2(16) + 11 = 32 + 11 = 43_{10}$.
    \item \textbf{Q.13 (C):} $2^8 - 1 = 255$ is the maximum unsigned integer value for 8 bits.
    \item \textbf{Q.23 (B):} \texttt{git commit} only captures snapshots into the local machine's \texttt{.git} folder.
\end{itemize}
\end{expbox}
