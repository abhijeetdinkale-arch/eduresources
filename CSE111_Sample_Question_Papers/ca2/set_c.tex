\begin{expbox}[Technical Solutions --- CA-2 Set C]
\begin{itemize}[leftmargin=5mm]
    \item \textbf{Q.5 (A):} $101111_2$ in 3-bit groups: $\underbrace{101}_{5}\;\underbrace{111}_{7} = (57)_8$.
    \item \textbf{Q.8 (B):} $0.101_2 = 1(0.5) + 0(0.25) + 1(0.125) = 0.625_{10}$.
    \item \textbf{Q.11 (A):} $(70)_8 = 7(8^1) + 0(8^0) = 56_{10}$.
    \item \textbf{Q.12 (B):} $(20)_{16} = 2(16^1) + 0(16^0) = 32_{10}$.
    \item \textbf{Q.19 (A):} \texttt{git fetch} downloads remote commits without merging, unlike \texttt{git pull} which merges automatically.
\end{itemize}
\end{expbox}
