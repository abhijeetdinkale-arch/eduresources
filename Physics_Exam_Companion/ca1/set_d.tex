\mcqitem{2}{If $\nabla \times \vec{A} = 0$, the vector field $\vec{A}$ is:}{Solenoidal}{Irrotational (Conservative)}{Non-conservative}{Rotational}
\mcqitem{3}{The electric potential $V$ satisfies Poisson's equation $\nabla^2 V = $}{0}{$-\rho/\epsilon_0$}{$\rho/\epsilon_0$}{$\mu_0 J$}
\mcqitem{4}{In a charge-free region ($\rho = 0$), Poisson's equation reduces to:}{Laplace's equation $\nabla^2 V = 0$}{Wave equation}{Continuity equation}{Helmholtz equation}
\mcqitem{5}{Gauss's Divergence Theorem converts:}{Surface integral to volume integral}{Line integral to surface integral}{Line integral to volume integral}{Double to single}
\mcqitem{6}{Stokes' Theorem relates:}{Line integral around closed curve and surface integral of curl}{Volume and surface}{Scalar and vector}{None}
\mcqitem{7}{The equation of continuity $\nabla \cdot \vec{J} + \frac{\partial \rho}{\partial t} = 0$ expresses:}{Conservation of charge}{Conservation of energy}{Conservation of momentum}{Conservation of mass}
\mcqitem{8}{Maxwell's displacement current density is given by:}{$\vec{J}_d = \frac{\partial \vec{D}}{\partial t}$}{$\vec{J}_d = \sigma \vec{E}$}{$\vec{J}_d = \vec{E} \times \vec{H}$}{$\vec{J}_d = \nabla \times \vec{H}$}

\mcqitem{10}{Faraday's law in differential form is:}{$\nabla \times \vec{E} = -\frac{\partial \vec{B}}{\partial t}$}{$\nabla \times \vec{B} = \mu_0 \vec{J}$}{$\nabla \cdot \vec{E} = \rho/\epsilon$}{$\nabla \cdot \vec{B} = 0$}
\mcqitem{11}{The Poynting vector $\vec{S} = \vec{E} \times \vec{H}$ represents:}{Rate of energy flow per unit area}{Electric charge density}{Magnetic energy}{Force on charge}
\mcqitem{12}{The velocity of electromagnetic waves in vacuum is given by:}{$c = 1/\sqrt{\mu_0 \epsilon_0}$}{$c = \sqrt{\mu_0 \epsilon_0}$}{$c = \mu_0 / \epsilon_0$}{$c = \epsilon_0 / \mu_0$}
\mcqitem{13}{The ratio of electric to magnetic field amplitudes $E/H$ in free space is:}{$377\,\Omega$ (Intrinsic Impedance $\eta_0$)}{$3 \times 10^8\,\Omega$}{$1\,\Omega$}{$0\,\Omega$}
\mcqitem{14}{The divergence of the curl of any vector field is:}{0}{1}{$\vec{A}$}{$\infty$}
\mcqitem{15}{The curl of the gradient of any scalar field is:}{0}{1}{$\nabla^2 \phi$}{$\phi$}
