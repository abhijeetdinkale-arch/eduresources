% Solution 1
\begin{solutionbox}[Solution to Question 1 (5 Marks)]
\textbf{Step 1: Definition of Continuity}
A function $f(x,y)$ is continuous at $(0,0)$ if and only if:
\[
\lim_{(x,y)\to (0,0)} f(x,y) = f(0,0) = 0
\]
\textbf{Step 2: Path Testing along $y = mx$}
Approach the origin along straight lines $y = mx$ where $m \neq -1$:
\[
\lim_{x\to 0} \frac{x^3 - (mx)^3}{x + mx} = \lim_{x\to 0} \frac{x^3(1 - m^3)}{x(1 + m)} = \lim_{x\to 0} x^2 \frac{1 - m^3}{1 + m} = \lim_{x\to 0} x^2 (1 - m + m^2) = 0
\]
This holds for any slope $m \neq -1$.

\textbf{Step 3: Path Testing approaching the singular line $y = -x$}
Let $y = -x + k x^2$:
\[
x^3 - y^3 = x^3 - (-x + kx^2)^3 = x^3 - (-x^3 + 3x^2(kx^2) - \dots) = 2x^3 - 3kx^4 + \dots
\]
\[
x + y = x + (-x + kx^2) = kx^2
\]
Then the limit becomes:
\[
\lim_{x\to 0} \frac{2x^3 - 3kx^4}{kx^2} = \lim_{x\to 0} \frac{2x}{k} = 0
\]
Furthermore, using polar coordinates $x = r\cos\theta, y = r\sin\theta$:
\[
\left|\frac{x^3 - y^3}{x+y}\right| = \left|\frac{r^3(\cos^3\theta - \sin^3\theta)}{r(\cos\theta + \sin\theta)}\right| = r^2 |\cos^2\theta - \cos\theta\sin\theta + \sin^2\theta| \le 2r^2 \to 0 \text{ as } r \to 0
\]
Since the limit as $(x,y)\to(0,0)$ exists and equals $0 = f(0,0)$, the function is \textbf{continuous at $(0,0)$}.

\begin{markingrubric}
\textbf{Mark Distribution:}
$\bullet$ Definition of multivariable continuity: \textbf{1.5 Marks}\\
$\bullet$ Evaluation along arbitrary paths / polar coordinates: \textbf{2 Marks}\\
$\bullet$ Proving $\lim = f(0,0) = 0$ and concluding continuity: \textbf{1.5 Marks}
\end{markingrubric}
\end{solutionbox}

% Solution 2
\begin{solutionbox}[Solution to Question 2 (5 Marks)]
Given $f(x,y,z) = x y^2 z^3$.

\textbf{Step 1: First Partial Derivatives}
\[
f_x = \frac{\partial f}{\partial x} = y^2 z^3, \quad f_y = \frac{\partial f}{\partial y} = 2x y z^3, \quad f_z = \frac{\partial f}{\partial z} = 3x y^2 z^2
\]
\textbf{Step 2: Second Partial Derivatives}
\begin{itemize}[leftmargin=6mm]
    \item $f_{xx} = \frac{\partial}{\partial x}(f_x) = \frac{\partial}{\partial x}(y^2 z^3) = \mathbf{0}$.
    \item $f_{xy} = \frac{\partial}{\partial y}(f_x) = \frac{\partial}{\partial y}(y^2 z^3) = \mathbf{2y z^3}$.
    \item $f_{yx} = \frac{\partial}{\partial x}(f_y) = \frac{\partial}{\partial x}(2x y z^3) = \mathbf{2y z^3}$.
    \item $f_{zx} = \frac{\partial}{\partial x}(f_z) = \frac{\partial}{\partial x}(3x y^2 z^2) = \mathbf{3y^2 z^2}$.
\end{itemize}
\textbf{Step 3: Verification}
Notice that $f_{xy} = 2y z^3 = f_{yx}$. The mixed partials are identically equal, verifying Schwarz's Theorem.

\begin{markingrubric}
\textbf{Mark Distribution:}
$\bullet$ First partial derivatives: \textbf{1.5 Marks}\\
$\bullet$ Computation of $f_{xx}, f_{zx}$: \textbf{1.5 Marks}\\
$\bullet$ Computation of $f_{xy}, f_{yx}$ and verification $f_{xy} = f_{yx}$: \textbf{2 Marks}
\end{markingrubric}
\end{solutionbox}

% Solution 3
\begin{solutionbox}[Solution to Question 3 (5 Marks)]
\textbf{Step 1: Problem Formulation}
Let $r$ be the radius and $h$ be the height of the cylinder.
Given volume $V$ is fixed:
\[
V = \pi r^2 h \implies h = \frac{V}{\pi r^2} \label{eq:cyl_vol}
\]
Since the cost per unit area is uniform (say $C$ per $\text{cm}^2$), minimizing cost is equivalent to minimizing the total surface area $S$:
\[
S = 2(\pi r^2) + 2\pi r h = 2\pi r^2 + 2\pi r \left(\frac{V}{\pi r^2}\right) = 2\pi r^2 + \frac{2V}{r}
\]
\textbf{Step 2: Finding Stationary Points}
Differentiate $S$ with respect to $r$:
\[
\frac{dS}{dr} = 4\pi r - \frac{2V}{r^2}
\]
Set $\frac{dS}{dr} = 0$:
\[
4\pi r = \frac{2V}{r^2} \implies r^3 = \frac{2V}{4\pi} = \frac{V}{2\pi} \implies \mathbf{r = \left(\frac{V}{2\pi}\right)^{1/3}}
\]
\textbf{Step 3: Second Derivative Test}
\[
\frac{d^2 S}{dr^2} = 4\pi + \frac{4V}{r^3} = 4\pi + \frac{4V}{V/(2\pi)} = 4\pi + 8\pi = 12\pi > 0
\]
Since $\frac{d^2S}{dr^2} > 0$, the surface area (and cost) is strictly **minimized**.

\textbf{Step 4: Optimal Dimensions}
Substitute $r$ into \eqref{eq:cyl_vol}:
\[
h = \frac{V}{\pi r^2} = \frac{2\pi r^3}{\pi r^2} = 2r
\]
Thus, the optimal dimensions are:
\[
\mathbf{r = \left(\frac{V}{2\pi}\right)^{1/3}, \quad h = 2r = 2\left(\frac{V}{2\pi}\right)^{1/3}}
\]
The height must equal the diameter ($h = 2r$) of the cylindrical base.

\begin{markingrubric}
\textbf{Mark Distribution:}
$\bullet$ Surface area objective function in single variable $r$: \textbf{1.5 Marks}\\
$\bullet$ First derivative and finding critical point $r = (V/2\pi)^{1/3}$: \textbf{1.5 Marks}\\
$\bullet$ Second derivative test confirming minimum: \textbf{1 Mark}\\
$\bullet$ Concluding optimal dimension relation $h = 2r$: \textbf{1 Mark}
\end{markingrubric}
\end{solutionbox}

% Solution 4
\begin{solutionbox}[Solution to Question 4 (5 Marks)]
Integral to evaluate:
\[
I = \int_0^1 \int_{y^2}^1 \int_0^{1-x} (y - 2z) \, dz \, dx \, dy
\]
\textbf{Step 1: Inner Integration with respect to $z$}
\[
\int_0^{1-x} (y - 2z) \, dz = \left[ y z - z^2 \right]_0^{1-x} = y(1 - x) - (1 - x)^2 = y - xy - (1 - 2x + x^2)
\]
\textbf{Step 2: Middle Integration with respect to $x$}
\[
\int_{y^2}^1 \left[ y(1 - x) - (1 - x)^2 \right] dx = \left[ -y\frac{(1-x)^2}{2} + \frac{(1-x)^3}{3} \right]_{y^2}^1
\]
At upper limit $x = 1$, $(1-1) = 0$.
At lower limit $x = y^2$:
\[
= 0 - \left[ -y\frac{(1-y^2)^2}{2} + \frac{(1-y^2)^3}{3} \right] = \frac{y(1-y^2)^2}{2} - \frac{(1-y^2)^3}{3}
\]
\textbf{Step 3: Outer Integration with respect to $y$}
\[
I = \int_0^1 \frac{y(1-y^2)^2}{2} \, dy - \int_0^1 \frac{(1-y^2)^3}{3} \, dy
\]
For the first integral, let $u = 1 - y^2 \implies du = -2y \, dy \implies y \, dy = -du/2$:
\[
I_1 = \frac{1}{2} \int_1^0 u^2 \left(-\frac{du}{2}\right) = \frac{1}{4} \int_0^1 u^2 \, du = \frac{1}{4}\left[\frac{1}{3}\right] = \frac{1}{12}
\]
For the second integral, expand $(1-y^2)^3 = 1 - 3y^2 + 3y^4 - y^6$:
\[
I_2 = \frac{1}{3} \int_0^1 (1 - 3y^2 + 3y^4 - y^6) \, dy = \frac{1}{3} \left[ 1 - 1 + \frac{3}{5} - \frac{1}{7} \right] = \frac{1}{3}\left[\frac{21 - 5}{35}\right] = \frac{16}{105}
\]
Subtract:
\[
I = I_1 - I_2 = \frac{1}{12} - \frac{16}{105} = \frac{35 - 64}{420} = \mathbf{-\frac{29}{420}}
\]

\begin{markingrubric}
\textbf{Mark Distribution:}
$\bullet$ Integration with respect to $z$: \textbf{1.5 Marks}\\
$\bullet$ Integration with respect to $x$: \textbf{1.5 Marks}\\
$\bullet$ Outer integral with respect to $y$ yielding $-29/420$: \textbf{2 Marks}
\end{markingrubric}
\end{solutionbox}

% Solution 5
\begin{solutionbox}[Solution to Question 5 (5 Marks)]
\textbf{Step 1: Polar Coordinate Transformation}
Let $x = r\cos\theta, y = r\sin\theta \implies x^2 + y^2 = r^2$, and $dA = r \, dr \, d\theta$.
\textbf{Step 2: Limits for the Second Quadrant of $x^2 + y^2 = 4$}
In the second quadrant ($x \le 0, y \ge 0$):
\[
\theta \in \left[\frac{\pi}{2}, \pi\right], \quad r \in [0, 2]
\]
\textbf{Step 3: Integral Evaluation}
\[
\iint_R (x^2 + y^2) \, dA = \int_{\pi/2}^\pi \int_0^2 (r^2) \cdot r \, dr \, d\theta = \left(\int_{\pi/2}^\pi d\theta\right) \left(\int_0^2 r^3 \, dr\right)
\]
\[
= \left(\pi - \frac{\pi}{2}\right) \left[ \frac{r^4}{4} \right]_0^2 = \left(\frac{\pi}{2}\right) \left(\frac{16}{4}\right) = \left(\frac{\pi}{2}\right)(4) = \mathbf{2\pi}
\]

\begin{markingrubric}
\textbf{Mark Distribution:}
$\bullet$ Polar transformation formulas and Jacobian $r$: \textbf{1.5 Marks}\\
$\bullet$ Setting second quadrant limits $\theta \in [\pi/2, \pi]$ and $r \in [0, 2]$: \textbf{1.5 Marks}\\
$\bullet$ Calculation yielding final answer $2\pi$: \textbf{2 Marks}
\end{markingrubric}
\end{solutionbox}

% Solution 6
\begin{solutionbox}[Solution to Question 6 (5 Marks)]
\textbf{Step 1: Setup Iterated Triple Integral}
Region $E$: $0 \le x \le 1$, $0 \le y \le 1$, $0 \le z \le x + y$.
\[
I = \int_0^1 \int_0^1 \int_0^{x+y} xy \, dz \, dy \, dx
\]
\textbf{Step 2: Integrate with respect to $z$}
\[
\int_0^{x+y} xy \, dz = xy [z]_0^{x+y} = xy(x+y) = x^2 y + x y^2
\]
\textbf{Step 3: Integrate with respect to $y$}
\[
\int_0^1 (x^2 y + x y^2) \, dy = \left[ x^2 \frac{y^2}{2} + x \frac{y^3}{3} \right]_0^1 = \frac{x^2}{2} + \frac{x}{3}
\]
\textbf{Step 4: Integrate with respect to $x$}
\[
I = \int_0^1 \left(\frac{x^2}{2} + \frac{x}{3}\right) \, dx = \left[ \frac{x^3}{6} + \frac{x^2}{6} \right]_0^1 = \frac{1}{6} + \frac{1}{6} = \mathbf{\frac{1}{3}}
\]

\begin{markingrubric}
\textbf{Mark Distribution:}
$\bullet$ Setting up iterated integral bounds: \textbf{1 Mark}\\
$\bullet$ First integration with respect to $z$: \textbf{1.5 Marks}\\
$\bullet$ Second integration with respect to $y$: \textbf{1 Mark}\\
$\bullet$ Final integration yielding $1/3$: \textbf{1.5 Marks}
\end{markingrubric}
\end{solutionbox}
