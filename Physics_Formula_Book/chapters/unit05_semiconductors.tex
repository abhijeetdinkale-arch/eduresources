\chapter{Unit 5: Semiconductor Physics \& Solid State}

\begin{formulabox}{1. Classical Drude-Lorentz Free Electron Theory}
\begin{itemize}[leftmargin=*,itemsep=2pt]
    \item \textbf{Drift Velocity \& Mobility:} $v_d = \mu E = \frac{e\tau}{m} E \implies \mu = \frac{e\tau}{m}$.
    \item \textbf{Electrical Conductivity:} $\sigma = n e \mu = \frac{n e^2 \tau}{m}$.
    \item \textbf{Wiedemann-Franz Law \& Lorentz Number:}
    \[
    \frac{K}{\sigma T} = L = \frac{\pi^2 k_B^2}{3 e^2} \approx 2.44 \times 10^{-8}\text{ W}\cdot\Omega/\text{K}^2
    \]
\end{itemize}
\end{formulabox}

\begin{formulabox}{2. Band Theory of Solids (Kronig-Penney Model)}
\begin{itemize}[leftmargin=*,itemsep=2pt]
    \item \textbf{Bloch Theorem:} $\psi_k(x) = e^{ikx} u_k(x)$, where $u_k(x+a) = u_k(x)$.
    \item \textbf{Kronig-Penney Dispersion Relation:}
    \[
    P\frac{\sin(\alpha a)}{\alpha a} + \cos(\alpha a) = \cos(ka), \quad \text{where } P = \frac{m V_0 b a}{\hbar^2}, \, \alpha = \sqrt{\frac{2mE}{\hbar^2}}
    \]
    $\bullet$ If $P \to 0$: Free electron parabola $E = \frac{\hbar^2 k^2}{2m}$.\\
    $\bullet$ If $P \to \infty$: Discrete atomic levels $\alpha a = n\pi \implies E_n = \frac{n^2 \pi^2 \hbar^2}{2ma^2}$.
    \item \textbf{Effective Mass of Electron:} $m^* = \frac{\hbar^2}{d^2 E / dk^2}$.
\end{itemize}
\end{formulabox}

\begin{formulabox}{3. Fermi-Dirac Distribution \& Carrier Statistics}
\begin{itemize}[leftmargin=*,itemsep=2pt]
    \item \textbf{Fermi-Dirac Occupation Probability:} $f(E) = \frac{1}{1 + \exp\left(\frac{E - E_F}{k_B T}\right)}$.
    \item \textbf{Intrinsic Carrier Concentration:} $n_i = \sqrt{N_c N_v} \exp\left(-\frac{E_g}{2k_B T}\right)$.
    \item \textbf{Intrinsic Fermi Level:} $E_F = \frac{E_c + E_v}{2} + \frac{3}{4}k_B T \ln\left(\frac{m_h^*}{m_e^*}\right) \approx \text{Midgap}$.
    \item \textbf{Extrinsic Fermi Levels:}
    \[
    \text{N-type: } E_F = E_c - k_B T\ln\left(\frac{N_c}{N_d}\right), \quad \text{P-type: } E_F = E_v + k_B T\ln\left(\frac{N_v}{N_a}\right)
    \]
    \item \textbf{Mass Action Law (Equilibrium):} $n \cdot p = n_i^2 = N_c N_v e^{-E_g/k_B T}$.
\end{itemize}
\end{formulabox}

\begin{formulabox}{4. Hall Effect \& Solar Cells}
\begin{itemize}[leftmargin=*,itemsep=2pt]
    \item \textbf{Hall Voltage:} $V_H = \frac{B I}{n q w} = \frac{R_H B I}{w}$.
    \item \textbf{Hall Coefficient:} $R_H = \frac{1}{n q} = \begin{cases} -\frac{1}{n e} & (\text{N-type electrons}) \\ +\frac{1}{p e} & (\text{P-type holes}) \end{cases}$
    \item \textbf{Hall Mobility:} $\mu_H = |R_H| \sigma$.
    \item \textbf{Solar Cell Fill Factor \& Conversion Efficiency:}
    \[
    \text{FF} = \frac{V_{\text{mp}} I_{\text{mp}}}{V_{\text{oc}} I_{\text{sc}}}, \quad \eta = \frac{P_{\text{max}}}{P_{\text{in}}} = \frac{\text{FF} \cdot V_{\text{oc}} \cdot I_{\text{sc}}}{P_{\text{in}} \cdot A_{\text{cell}}}
    \]
\end{itemize}
\end{formulabox}
