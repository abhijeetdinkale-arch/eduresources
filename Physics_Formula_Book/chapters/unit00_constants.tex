\begin{formulabox}{Crucial Energy and Dimensional Conversion Formulas}
\begin{itemize}[leftmargin=*,itemsep=2pt]
    \item \textbf{Electron-Volt to Joules:} $1\text{ eV} = 1.60218 \times 10^{-19}\text{ J} \iff 1\text{ J} = 6.2415 \times 10^{18}\text{ eV}$.
    \item \textbf{Photon Energy vs Wavelength:} $E = h\nu = \frac{hc}{\lambda} = \frac{1240\text{ eV}\cdot\text{nm}}{\lambda\text{ (in nm)}} = \frac{12400\text{ eV}\cdot\text{\AA}}{\lambda\text{ (in \AA)}}$.
    \item \textbf{Thermal Energy at Room Temperature ($T = 300\text{ K}$):} $k_B T \approx 0.02585\text{ eV} \approx 26\text{ meV} \approx 4.14 \times 10^{-21}\text{ J}$.
    \item \textbf{de Broglie Wavelength of Electron accelerated through $V$ Volts:}
    \[
    \lambda_e = \frac{h}{\sqrt{2m_e e V}} = \frac{1.227}{\sqrt{V}}\text{ nm} = \sqrt{\frac{150}{V}}\text{ \AA}
    \]
\end{itemize}
\end{formulabox}
