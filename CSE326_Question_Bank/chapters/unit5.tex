\section{Section 3: Conceptual Theory \& Coding Questions}
\noindent\textbf{Q.51 [Theory --- 5 Marks]} Explain Event Delegation in the JavaScript DOM. Why is it more performant than attaching individual listeners?

\begin{qbsolutionbox}[Answer to Q.51]
\textbf{Event Delegation} is an event handling pattern where a single event listener is attached to a parent element rather than attaching separate listeners to each child element.
\begin{itemize}[leftmargin=*]
  \item \textbf{Mechanism}: It leverages \textbf{Event Bubbling}, where events triggered on child nodes propagate upward through parent ancestors. The parent handler inspects \texttt{event.target} to identify which child originated the action.
  \item \textbf{Performance Benefits}:
  \begin{enumerate}
    \item \textbf{Memory Efficiency}: Conserves memory by creating 1 listener instead of hundreds (e.g. on a 500-row table).
    \item \textbf{Dynamic DOM Elements}: Automatically handles newly added elements without needing to bind new listeners after AJAX/DOM inserts.
  \end{enumerate}
\end{itemize}
\end{qbsolutionbox}

\noindent\textbf{Q.52 [Coding --- 5 Marks]} Write a complete JavaScript email validation script using Regular Expressions that validates input on form submission.

\begin{qbsolutionbox}[Answer to Q.52]
\begin{lstlisting}[language=JavaScript]
function validateEmailForm(event) {
  event.preventDefault(); // Stop native submission
  
  const emailField = document.getElementById("emailInput").value.trim();
  const emailRegex = /^[^ ]+@[^ ]+\.[a-z]{2,3}$/;
  const errorDisplay = document.getElementById("errorMsg");

  if (emailField === "") {
    errorDisplay.textContent = "Email field cannot be empty!";
    return false;
  }
  if (!emailRegex.test(emailField)) {
    errorDisplay.textContent = "Invalid email format (e.g. user@domain.com)!";
    return false;
  }

  errorDisplay.textContent = "";
  alert("Email validated successfully!");
  return true;
}
\end{lstlisting}
\end{qbsolutionbox}
