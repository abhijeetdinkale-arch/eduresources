\section{Section 3: Conceptual Theory \& Coding Questions}
\noindent\textbf{Q.51 [Theory --- 5 Marks]} Differentiate between \texttt{<article>}, \texttt{<section>}, and \texttt{<div>}. Explain when each should be used.

\begin{qbsolutionbox}[Answer to Q.51]
\begin{itemize}[leftmargin=*]
  \item \textbf{<article>}: A self-contained, independently distributable composition that makes sense in isolation (e.g. blog post, product card, forum comment, newspaper story).
  \item \textbf{<section>}: A thematic grouping of content, typically introduced with its own heading, forming a distinct chapter or section of a document.
  \item \textbf{<div>}: A non-semantic generic block container used purely for CSS styling or JavaScript hooks when no semantic HTML5 element applies.
\end{itemize}
\end{qbsolutionbox}

\noindent\textbf{Q.52 [Coding --- 5 Marks]} Write the complete HTML5 markup for an accessible registration form with Name, Email, Password, Course selection, and Submit button.

\begin{qbsolutionbox}[Answer to Q.52]
\begin{lstlisting}[language=HTML]
<form action="register.php" method="POST">
  <fieldset>
    <legend>Student Registration</legend>
    <p>
      <label for="uName">Full Name:</label>
      <input type="text" id="uName" name="username" required>
    </p>
    <p>
      <label for="uEmail">Email Address:</label>
      <input type="email" id="uEmail" name="useremail" required>
    </p>
    <p>
      <label for="uPass">Password (Min 8 chars):</label>
      <input type="password" id="uPass" name="password" minlength="8" required>
    </p>
    <p>
      <label for="uCourse">Course:</label>
      <select id="uCourse" name="course">
        <option value="cse" selected>CSE</option>
        <option value="it">IT</option>
      </select>
    </p>
    <button type="submit">Submit Registration</button>
  </fieldset>
</form>
\end{lstlisting}
\end{qbsolutionbox}
