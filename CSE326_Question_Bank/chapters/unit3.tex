\section{Section 3: Conceptual Theory \& Coding Questions}
\noindent\textbf{Q.51 [Theory --- 5 Marks]} Explain the CSS Specificity calculation model $(a, b, c)$ and demonstrate with three selector examples.

\begin{qbsolutionbox}[Answer to Q.51]
CSS Specificity is calculated as a weight triple $(a, b, c)$:
\begin{itemize}[leftmargin=*]
  \item \textbf{a (IDs)}: Number of ID selectors (\texttt{\#header}) $\to$ Weight = $100$.
  \item \textbf{b (Classes, Attributes, Pseudo-classes)}: Number of classes (\texttt{.btn}), attributes (\texttt{[type="text"]}), and pseudo-classes (\texttt{:hover}) $\to$ Weight = $10$.
  \item \textbf{c (Elements, Pseudo-elements)}: Number of type selectors (\texttt{p}, \texttt{div}) and pseudo-elements (\texttt{::before}) $\to$ Weight = $1$.
\end{itemize}
\textbf{Examples}:
\begin{enumerate}
  \item \texttt{div\#main p} $\to 1\text{ ID} + 0\text{ Classes} + 2\text{ Elements} = \mathbf{(1, 0, 2)}$.
  \item \texttt{header nav.menu ul li.active a:hover} $\to 0\text{ IDs} + 3\text{ Classes/Pseudos} + 4\text{ Elements} = \mathbf{(0, 3, 4)}$.
  \item \texttt{\#main} beats \texttt{header nav.menu ul li.active a:hover} because $a=1 > a=0$.
\end{enumerate}
\end{qbsolutionbox}

\noindent\textbf{Q.52 [Coding --- 5 Marks]} Write CSS rules implementing a responsive CSS Box Model container with background image cover, border-radius, and auto margin centering.

\begin{qbsolutionbox}[Answer to Q.52]
\begin{lstlisting}[language=HTML]
<style>
  * {
    box-sizing: border-box;
  }
  .profile-card {
    width: 320px;
    margin: 30px auto;
    padding: 20px;
    background: #ffffff url("pattern.png") no-repeat center / cover;
    border: 2px solid #0066cc;
    border-radius: 12px;
    text-align: center;
    box-shadow: 0 4px 10px rgba(0,0,0,0.15);
  }
</style>
<div class="profile-card">
  <h2>John Doe</h2>
  <p>Software Engineer</p>
</div>
\end{lstlisting}
\end{qbsolutionbox}
