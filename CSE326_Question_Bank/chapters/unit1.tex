\section{Section 3: Conceptual Theory \& Coding Questions}
\noindent\textbf{Q.51 [Theory --- 5 Marks]} Differentiate between Block-level elements and Inline elements in HTML5. Provide three examples of each.

\begin{qbsolutionbox}[Answer to Q.51]
\begin{itemize}[leftmargin=*]
  \item \textbf{Block-level Elements}: Always begin on a new line and expand horizontally across 100\% of the parent container's available width. They can contain other block-level or inline elements. Examples: \texttt{<div>}, \texttt{<p>}, \texttt{<h1>-<h6>}, \texttt{<ul>}, \texttt{<section>}, \texttt{<main>}.
  \item \textbf{Inline Elements}: Flow within the natural line of text without forcing a line break. They occupy only the exact width required by their content. Examples: \texttt{<span>}, \texttt{<a>}, \texttt{<strong>}, \texttt{<em>}, \texttt{<img>}, \texttt{<label>}.
\end{itemize}
\end{qbsolutionbox}

\noindent\textbf{Q.52 [Coding --- 5 Marks]} Write a valid HTML5 document structure containing meta viewport and charset configurations.

\begin{qbsolutionbox}[Answer to Q.52]
\begin{lstlisting}[language=HTML]
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Course Portal</title>
</head>
<body>
  <h1>Welcome to Web Technologies</h1>
</body>
</html>
\end{lstlisting}
\end{qbsolutionbox}
