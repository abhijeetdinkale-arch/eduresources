\noindent\textbf{Q.52 [Coding --- 5 Marks]} Write a complete JavaScript module that saves a user preferences object (theme and fontSize) into \texttt{localStorage} and restores it on page load.

\begin{qbsolutionbox}[Answer to Q.52]
\begin{lstlisting}[language=JavaScript]
// Save preferences to localStorage
function savePreferences(theme, fontSize) {
  const prefs = {
    theme: theme,
    fontSize: fontSize,
    updatedAt: new Date().toISOString()
  };
  localStorage.setItem("userPreferences", JSON.stringify(prefs));
  console.log("Preferences saved successfully.");
}

// Load and apply preferences from localStorage
function loadPreferences() {
  const rawData = localStorage.getItem("userPreferences");
  if (!rawData) return; // Use defaults if empty

  try {
    const prefs = JSON.parse(rawData);
    document.body.className = prefs.theme; // Apply theme class
    document.body.style.fontSize = prefs.fontSize; // Apply font size
    console.log("Preferences applied:", prefs);
  } catch (err) {
    console.error("Error loading preferences:", err);
  }
}

// Attach to page load event
window.addEventListener("DOMContentLoaded", loadPreferences);
\end{lstlisting}
\end{qbsolutionbox}
