\noindent Welcome to \textbf{MEC136: Engineering Graphics \& Computer Aided Drafting}. This textbook has been authored from the ground up to serve as the definitive, university-level curriculum companion, laboratory drafting manual, and high-yield exam preparation guide.

\section*{Strategic Exam Preparation Paradigms}

\begin{tcolorbox}[enhanced, colback=boxtealbg, colframe=brandteal, title={\textbf{Scenario A: Full Semester Study Strategy (Maximum Marks + Mastery)}}, boxrule=1.2pt]
\begin{enumerate}
    \item \textbf{Understand Core Concepts:} Read the topic-by-topic theory, focusing on physical principles (e.g., Similar Triangles in Diagonal Scales, Rotating Line Method for Lines, Glass Box concept for Orthographic Projections).
    \item \textbf{Master Manual Drafting Procedures:} Follow step-by-step geometric construction steps for line projections, sectional views, and surface developments.
    \item \textbf{Learn AutoCAD Workflows:} Memorize command shortcuts, options, and key toggles (\texttt{UNITS}, \texttt{LIMITS}, \texttt{OSNAP F3}, \texttt{ORTHO F8}, \texttt{HATCH}, \texttt{EXTRUDE}).
    \item \textbf{Solve Graded Exercises:} Work through Level 1 to Level 4 practice problems at the end of each unit.
\end{enumerate}
\end{tcolorbox}

\begin{tcolorbox}[enhanced, colback=boxamberbg, colframe=brandamber, title={\textbf{Scenario B: Limited Time / Mid-Term Strategy (High-Yield Focus)}}, boxrule=1.2pt]
\begin{enumerate}
    \item \textbf{Focus on [HIGH PRIORITY] Topics:} First Angle vs. Third Angle view placement, Line Projections inclined to both planes, Isometric Scale factor derivation ($0.816$), Sectional hatching exceptions (ribs/fasteners).
    \item \textbf{Review Key Formulas \& Derivations:} Representative Fraction ($R.F.$), Isometric Scale derivation, Sector Angle ($\theta = \frac{r}{L} \times 360^\circ$).
    \item \textbf{Solve Exam Problems:} Practice 5 solved numerical/construction problems per unit.
\end{enumerate}
\end{tcolorbox}

\begin{tcolorbox}[enhanced, colback=boxrosebg, colframe=boxroseborder, title={\textbf{Scenario C: Last-Day Night Before Exam Strategy (Rapid Revision)}}, boxrule=1.2pt]
\begin{enumerate}
    \item Read \textbf{Chapter 7: Master Revision Sheet} (Formula summary, Definition index, AutoCAD Cheat Sheet).
    \item Review \textbf{Common Student Pitfalls} boxes in Chapters 1--6.
    \item Attempt \textbf{Chapter 9: Mock Examination} under timed conditions.
\end{enumerate}
\end{tcolorbox}
